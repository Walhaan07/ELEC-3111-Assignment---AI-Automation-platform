One command starts everything

package.json — root 
{
  "type": "module",
  "workspaces": ["apps/*", "packages/*"],
  "engines": { "node": ">=22" },
  "scripts": {
    "doctor":  "node scripts/doctor.js",
    "predev":  "npm run doctor",
    "dev":     "docker compose up -d --wait && npm run dev --workspaces --if-present",
    "migrate": "psql $DATABASE_URL -f schema.sql && psql $DATABASE_URL -f seed.sql",
    "test":    "vitest run --coverage",
    "e2e":     "playwright test",
    "lint":    "eslint . && tsc --noEmit",
    "format":  "prettier --write .",
    "secrets": "gitleaks detect --no-git --redact"
  }
}
$ git clone … && cd ai-automation $ cp .env.example .env $ npm install $ npm run migrate $ npm run dev # editor 5173 · api 5678 · postgres 5432

Five commands, ten minutes, no questions asked of anybody. That is the target. --wait makes docker block until the healthcheck passes, and predev makes npm run the doctor first, automatically, every single time.

The doctor — forty lines that answer "why won't it start?"

Eight people on four operating systems will hit the same six problems. Write the answers down once, in code, and stop answering them in the group chat.

scripts/doctor.js 
import { execSync } from 'node:child_process';
import fs from 'node:fs';
import pg from 'pg';

const GREEN = '\x1b[32m', RED = '\x1b[31m', OFF = '\x1b[0m';
const results = [];

const check = (name, fn, fix) => {
  try { results.push(['ok', name, fn() ?? '']); }
  catch (e) { results.push(['fail', name, `${e.message}  →  ${fix}`]); }
};

check('Node 22 or newer', () => {
  const major = Number(process.versions.node.split('.')[0]);
  if (major < 22) throw new Error(`found v${process.versions.node}`);
  return `v${process.versions.node}`;
}, 'install Node 22 (nvm install 22), then reopen the terminal');

check('Docker is running',
  () => execSync('docker info --format "{{.ServerVersion}}"',
                 { stdio: ['ignore', 'pipe', 'ignore'] }).toString().trim(),
  'start Docker Desktop and wait for the whale to stop animating');

check('.env exists',
  () => { if (!fs.existsSync('.env')) throw new Error('missing'); return ''; },
  'run: cp .env.example .env');

for (const key of ['DATABASE_URL', 'BASE_URL', 'ENCRYPTION_KEY']) {
  check(`${key} is set`, () => {
    if (!process.env[key]) throw new Error('empty');
    if (key === 'ENCRYPTION_KEY' && !/^[0-9a-f]{64}$/i.test(process.env[key]))
      throw new Error('must be 64 hex characters');
    return '';
  }, `fill ${key} in .env — see .env.example`);
}

const db = new pg.Pool({ connectionString: process.env.DATABASE_URL,
                         connectionTimeoutMillis: 3000 });
try {
  const { rows } = await db.query("SELECT count(*)::int AS n FROM information_schema.tables "
                                + "WHERE table_name IN ('workflows','executions')");
  results.push(rows[0].n === 2
    ? ['ok',   'Tables exist', 'workflows, executions']
    : ['fail', 'Tables exist', 'not migrated  →  run: npm run migrate']);
} catch (e) {
  results.push(['fail', 'Postgres answers', `${e.message}  →  run: docker compose up -d`]);
} finally {
  await db.end();
}

for (const [state, name, detail] of results)
  console.log(`${state === 'ok' ? `${GREEN} ok ${OFF}` : `${RED}FAIL${OFF}`} `
            + `${name.padEnd(22)} ${detail}`);

const failed = results.filter(r => r[0] === 'fail').length;
console.log(failed ? `\n${failed} problem(s) — fix the arrows above.` : '\nEverything is ready.');
process.exit(failed ? 1 : 0);      // a non-zero exit is what stops `npm run dev` continuing
