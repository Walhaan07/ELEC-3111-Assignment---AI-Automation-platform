Level 1 — the engine, in milliseconds

Test the happy path once, then test every guard you wrote in Sec. 2.1b. Those six tests are the proof that "robust" is not just a word in the report.

packages/engine/engine.test.js 
import { test, expect } from 'vitest';
import { runWorkflow, validateWorkflow } from './engine.js';
import { WorkflowError } from './errors.js';

const fake = (name, out, group = 'action') => ({
  description: { name, group, properties: [] },
  async execute() { return typeof out === 'function' ? out() : out; },
});
const wf = (nodes, connections) => ({ nodes, connections });

test('runs nodes in order and passes items along', async () => {
  const types = { start: fake('start', [[{ json: { a: 1 } }]], 'trigger'),
                  next:  fake('next',  [[{ json: { b: 2 } }]]) };
  const { results, log } = await runWorkflow(
    wf([{ name: 'S', type: 'start', parameters: {} },
        { name: 'N', type: 'next',  parameters: {} }], { S: [['N']] }), types);
  expect(results.N[0][0].json).toEqual({ b: 2 });
  expect(log.filter(l => l.level === 'done')).toHaveLength(2);
});

test('an empty branch stops everything after it', async () => {
  const types = { start: fake('start', [[]], 'trigger'), never: fake('never', [[{ json: {} }]]) };
  const { results } = await runWorkflow(
    wf([{ name: 'S', type: 'start', parameters: {} },
        { name: 'X', type: 'never', parameters: {} }], { S: [['X']] }), types);
  expect(results.X).toBeUndefined();          // it never ran — and that is correct
});

test('a misspelled target is refused before anything runs', () => {
  const types = { start: fake('start', [[{ json: {} }]], 'trigger') };
  expect(() => validateWorkflow(
    wf([{ name: 'S', type: 'start', parameters: {} }], { S: [['Wether']] }), types))
    .toThrow(/missing node "Wether"/);
});

test('a loop is stopped by the step limit, not by the laptop', async () => {
  const types = { start: fake('start', [[{ json: {} }]], 'trigger'),
                  echo:  fake('echo',  [[{ json: {} }]]) };
  await expect(runWorkflow(
    wf([{ name: 'S', type: 'start', parameters: {} },
        { name: 'E', type: 'echo',  parameters: {} }], { S: [['E']], E: [['E']] }),
    types, { maxSteps: 25 })).rejects.toThrow(/loops/);
});

test('a hanging node is cut off and names itself', async () => {
  const types = { start: fake('start', [[{ json: {} }]], 'trigger'),
                  slow:  { description: { name: 'slow', group: 'action', properties: [] },
                           execute: () => new Promise(() => {}) } };     // never resolves
  await expect(runWorkflow(
    wf([{ name: 'S', type: 'start', parameters: {} },
        { name: 'Slow', type: 'slow', parameters: {} }], { S: [['Slow']] }),
    types, { nodeTimeoutMs: 40 })).rejects.toThrow(/"Slow" ran longer than/);
});

test('continueOnFail carries the error downstream instead of stopping', async () => {
  const types = { start: fake('start', [[{ json: {} }]], 'trigger'),
                  boom:  { description: { name: 'boom', group: 'action', properties: [] },
                           execute() { throw new Error('kaboom'); } },
                  after: fake('after', [[{ json: { reached: true } }]]) };
  const { results } = await runWorkflow(
    wf([{ name: 'S', type: 'start', parameters: {} },
        { name: 'B', type: 'boom',  parameters: {}, continueOnFail: true },
        { name: 'A', type: 'after', parameters: {} }],
       { S: [['B']], B: [['A']] }), types);
  expect(results.B[0][0].error).toMatch(/kaboom/);
  expect(results.A[0][0].json.reached).toBe(true);
});

The one test that will save the editor team a week

apps/editor/src/convert.test.ts 
import { test, expect } from 'vitest';
import { toReactFlow, fromReactFlow } from './convert';

const sample = {
  name: 'demo',
  nodes: [{ name: 'A', type: 'x', parameters: {}, position: { x: 0, y: 0 } },
          { name: 'B', type: 'y', parameters: {}, position: { x: 200, y: 0 } },
          { name: 'C', type: 'y', parameters: {}, position: { x: 200, y: 120 } }],
  connections: { A: [['B'], ['C']] },        // branch 0 → B, branch 1 → C (an IF node)
};

test('a workflow survives a round trip through React Flow unchanged', () => {
  const { nodes, edges } = toReactFlow(sample);
  expect(fromReactFlow(nodes, edges).connections).toEqual(sample.connections);
});

test('an edge to a deleted node is dropped, not crashed on', () => {
  const broken = { ...sample, connections: { A: [['B'], ['GHOST']] } };
  expect(toReactFlow(broken).edges).toHaveLength(1);
});

Level 2 — a node against a fake Google, with no internet

packages/nodes/google/sheets.test.js 
import { test, expect, beforeAll, afterAll } from 'vitest';
import http from 'node:http';

let server, calls = [], nextStatus = 200;

beforeAll(() => new Promise(done => {
  server = http.createServer((req, res) => {
    let body = ''; req.on('data', c => body += c);
    req.on('end', () => {
      calls.push({ url: req.url, method: req.method, body: JSON.parse(body || '{}') });
      if (nextStatus !== 200) {                       // let a test ask for a failure
        res.writeHead(nextStatus, { 'retry-after': '0' }); return res.end('{}');
      }
      res.writeHead(200, { 'content-type': 'application/json' });
      res.end(JSON.stringify({ values: [['name', 'email']],
                               updates: { updatedRows: 1, updatedRange: 'Orders!A18' } }));
    });
  }).listen(4010, done);
}));
afterAll(() => server.close());

test('append sends one row, in header order, to the right range', async () => {
  await sheetsAppend.execute(makeCtx({
    base: 'http://localhost:4010',
    documentId: 'https://docs.google.com/spreadsheets/d/ABC123/edit',
    sheetName: 'Orders', columns: { email: 'a@example.com', name: 'Alice' },
  }));
  const write = calls.at(-1);
  expect(write.method).toBe('POST');
  expect(write.url).toContain('/ABC123/values/');
  expect(write.body.values).toEqual([['Alice', 'a@example.com']]);   // sheet order, not object order
});

test('three items become ONE append request', async () => {
  calls = [];
  await sheetsAppend.execute(makeCtx({ base: 'http://localhost:4010', items: 3, /* … */ }));
  expect(calls.filter(c => c.method === 'POST')).toHaveLength(1);
});

test('a 429 is retried, not thrown', async () => {
  calls = []; nextStatus = 429;
  setTimeout(() => { nextStatus = 200; }, 30);        // recover after the first retry
  await expect(sheetsAppend.execute(makeCtx({ base: 'http://localhost:4010' })))
    .resolves.toBeDefined();
  expect(calls.length).toBeGreaterThan(1);
});

test('a sheet name with an apostrophe is quoted properly', () => {
  expect(quoteSheetName("Bob's list")).toBe("'Bob''s list'");
});

Thirty lines, and every Google node gets the same treatment. No credentials, no quota, no network — so it runs on every push in about two seconds.
