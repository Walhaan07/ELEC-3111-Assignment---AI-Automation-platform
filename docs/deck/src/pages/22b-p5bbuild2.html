Gmail — build the message by hand, then base64url it

This is the only genuinely fiddly node. Knowing that in advance stops it feeling like failure. Three things bite: non-English characters in the subject, addresses that are not addresses, and attachments over Gmail's limit.

packages/nodes/google/gmail.js 
const MAX_MESSAGE = 35 * 1024 * 1024;          // Gmail's hard limit for a send

// "Résumé" in a Subject: header is illegal ASCII. RFC 2047 is how you smuggle it through.
const encodeHeader = (s) => /^[\x20-\x7E]*$/.test(s)
  ? s
  : `=?UTF-8?B?${Buffer.from(s, 'utf8').toString('base64')}?=`;

const isEmail = (s) => /^[^\s@,;]+@[^\s@,;]+\.[^\s@,;]+$/.test(String(s).trim());

function addressList(value, field) {
  const list = String(value ?? '').split(',').map(s => s.trim()).filter(Boolean);
  if (list.length === 0) throw new Error(`${field} is empty`);
  const bad = list.filter(a => !isEmail(a));
  if (bad.length) throw new Error(`${field} contains an invalid address: ${bad[0]}`);
  return list.join(', ');
}

export function buildMime({ to, cc, subject, html, text, attachments = [], inReplyTo }) {
  const b = 'x' + crypto.randomUUID();
  const headers = [
    `To: ${addressList(to, 'To')}`,
    cc ? `Cc: ${addressList(cc, 'Cc')}` : null,
    `Subject: ${encodeHeader(subject ?? '(no subject)')}`,
    inReplyTo ? `In-Reply-To: ${inReplyTo}` : null,          // makes it a reply, not a new thread
    inReplyTo ? `References: ${inReplyTo}` : null,
    'MIME-Version: 1.0',
    `Content-Type: multipart/mixed; boundary="${b}"`,
  ].filter(Boolean).join('\r\n');

  const bodyPart =
    `--${b}\r\nContent-Type: text/html; charset="UTF-8"\r\n`
    + `Content-Transfer-Encoding: base64\r\n\r\n`
    + `${Buffer.from(html ?? text ?? '', 'utf8').toString('base64')}\r\n`;

  const fileParts = attachments.map(a =>
    `--${b}\r\nContent-Type: ${a.mimeType || 'application/octet-stream'}\r\n`
    + `Content-Disposition: attachment; filename="${a.fileName.replace(/"/g, '')}"\r\n`
    + `Content-Transfer-Encoding: base64\r\n\r\n${a.data}\r\n`).join('');

  const mime = `${headers}\r\n\r\n${bodyPart}${fileParts}--${b}--\r\n`;
  if (Buffer.byteLength(mime) > MAX_MESSAGE)
    throw new Error('That message is over Gmail’s 35 MB limit — put the file in Drive and link it');
  return mime;
}

const base64url = (s) => Buffer.from(s).toString('base64')
  .replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');

// dry-run is a real feature: it renders and validates without sending, so the whole
// workflow can be tested a hundred times without a hundred emails to a real person
if (ctx.getNodeParameter('dryRun', i, false)) {
  out.push({ json: { dryRun: true, preview: mime.slice(0, 400) }, pairedItem: i });
} else {
  const sent = await googleRequest(ctx.credentialId,
    'https://gmail.googleapis.com/gmail/v1/users/me/messages/send',
    { method: 'POST', body: JSON.stringify({ raw: base64url(mime),
                                             threadId: ctx.getNodeParameter('threadId', i, undefined) }) });
  out.push({ json: { messageId: sent.id, threadId: sent.threadId }, pairedItem: i });
}

The Code node — a thread we can stop, and starve

Two limits, not one. A timeout stops slow code; only a memory limit stops const a = []; while (true) a.push('x') from taking the server with it.

packages/nodes/core/code.js 
import { Worker } from 'node:worker_threads';

const WORKER = `
  const vm = require('node:vm');
  const { parentPort, workerData } = require('node:worker_threads');
  const logs = [];
  const sandbox = {
    items: structuredClone(workerData.items),      // a copy: user code cannot mutate ours
    console: { log: (...a) => logs.push(a.map(String).join(' ')).slice(0, 100) },
    JSON, Math, Date, String, Number, Boolean, Array, Object, RegExp,
  };
  try {
    const result = vm.runInNewContext(workerData.code, vm.createContext(sandbox),
                                      { timeout: workerData.timeoutMs, displayErrors: true });
    parentPort.postMessage({ ok: true, result, logs });
  } catch (e) {
    parentPort.postMessage({ ok: false, error: e && e.message ? e.message : String(e), logs });
  }
`;

export function runUserCode(code, items, { timeoutMs = 5000, memoryMb = 64 } = {}) {
  return new Promise((resolve, reject) => {
    if (typeof code !== 'string' || code.trim() === '')
      return reject(new Error('The Code node is empty'));

    const worker = new Worker(WORKER, {
      eval: true,
      workerData: { code, items, timeoutMs },
      resourceLimits: { maxOldGenerationSizeMb: memoryMb, maxYoungGenerationSizeMb: 16 },
      stdout: true, stderr: true,          // user code cannot spam our server log
    });

    let settled = false;
    const finish = (fn, value) => { if (settled) return; settled = true;
                                    clearTimeout(timer); worker.terminate(); fn(value); };

    // the vm timeout can be beaten by a tight async loop, so the thread gets a hard deadline too
    const timer = setTimeout(
      () => finish(reject, new Error(`Code node timed out after ${timeoutMs} ms`)),
      timeoutMs + 500);

    worker.on('message', (m) => {
      if (!m.ok) return finish(reject, new Error(`Code node: ${m.error}`));
      if (!Array.isArray(m.result))
        return finish(reject, new Error('Code must return an array of items, e.g. return items;'));
      finish(resolve, { items: m.result.map(x => ('json' in (x ?? {}) ? x : { json: x })),
                        logs: m.logs });
    });
    worker.on('error', (e) => finish(reject,
      /heap out of memory|Worker terminated/i.test(e.message)
        ? new Error(`Code node used more than ${memoryMb} MB and was stopped`)
        : e));
    worker.on('exit', (code) => { if (!settled && code !== 0)
      finish(reject, new Error(`Code node exited unexpectedly (${code})`)); });
  });
}

The AI node — one request, and everything that can go wrong with it

packages/nodes/ai/anthropic.js 
const MODELS = ['claude-opus-5', 'claude-sonnet-5', 'claude-haiku-4-5-20251001'];

async function ask(ctx, i) {
  const prompt = String(ctx.getNodeParameter('userPrompt', i));
  if (prompt.length > 20_000) throw new Error('Prompt is too long — trim the data you send in');

  const body = {
    model:      ctx.getNodeParameter('model', i, MODELS[1]),
    max_tokens: Math.min(Math.max(ctx.getNodeParameter('maxTokens', i, 512), 1), 4096),
    system:     ctx.getNodeParameter('systemPrompt', i, ''),
    messages:  [{ role: 'user', content: prompt }],
  };

  for (let attempt = 0; attempt < 4; attempt++) {
    const res = await fetch(`${ctx.credentials.baseUrl ?? 'https://api.anthropic.com'}/v1/messages`, {
      method: 'POST', signal: AbortSignal.timeout(60_000),
      headers: { 'x-api-key': ctx.credentials.apiKey, 'anthropic-version': '2023-06-01',
                 'content-type': 'application/json' },
      body: JSON.stringify(body),
    });

    if (res.status === 429 || res.status === 529 || res.status >= 500) {
      const wait = Number(res.headers.get('retry-after')) * 1000 || 2 ** attempt * 1000;
      ctx.logger.info(`model busy (${res.status}) — waiting ${wait} ms`);
      await new Promise(r => setTimeout(r, wait));
      continue;
    }
    const data = await res.json();
    if (!res.ok) throw new Error(`AI ${res.status}: ${data.error?.message ?? 'unknown error'}`);
    if (data.stop_reason === 'max_tokens')
      ctx.logger.info('answer was cut off — raise Max tokens');

    const text = (data.content ?? []).filter(c => c.type === 'text').map(c => c.text).join('');
    return { text, usage: data.usage, model: data.model, stopReason: data.stop_reason };
  }
  throw new Error('The model stayed busy after four attempts');
}

// asking for JSON and getting prose is the classic failure. Ask, then check, then repair once.
export function parseJsonAnswer(text, expectedKeys = []) {
  const block = text.match(/```(?:json)?\s*([\s\S]*?)```/)?.[1] ?? text;
  const slice = block.slice(block.indexOf('{'), block.lastIndexOf('}') + 1);
  let parsed;
  try { parsed = JSON.parse(slice); }
  catch { throw new Error(`The model did not return JSON. It said: ${text.slice(0, 120)}…`); }
  const missing = expectedKeys.filter(k => !(k in parsed));
  if (missing.length) throw new Error(`The answer is missing: ${missing.join(', ')}`);
  return parsed;
}
