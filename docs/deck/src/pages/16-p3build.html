This one description is simultaneously the form in the sidebar, the validation rules, and the documentation. Every field you add here appears in the UI with no UI work at all.

import { WorkflowError } from '../../engine/errors.js';

export const httpRequestNode = {
  description: {
    name: 'httpRequest', displayName: 'HTTP Request', group: 'action',
    icon: 'globe', colour: '#0d72b4', inputs: ['main'], outputs: ['main'],
    properties: [
      { displayName: 'Method', name: 'method', type: 'options', default: 'GET', required: true,
        options: ['GET','POST','PUT','PATCH','DELETE'].map(v => ({ name: v, value: v })) },

      { displayName: 'URL', name: 'url', type: 'string', default: '', required: true,
        placeholder: 'https://api.example.com/orders',
        validate: { pattern: '^https?://', message: 'must start with http:// or https://' } },

      { displayName: 'Send headers', name: 'sendHeaders', type: 'boolean', default: false },
      { displayName: 'Headers', name: 'headers', type: 'json', default: '{}',
        displayOptions: { show: { sendHeaders: [true] } } },

      { displayName: 'Body', name: 'body', type: 'json', default: '{}',
        displayOptions: { show: { method: ['POST','PUT','PATCH'] } } },   // hidden otherwise

      { displayName: 'Timeout (ms)', name: 'timeout', type: 'number', default: 30000,
        validate: { min: 100, max: 120000 } },
      { displayName: 'Retries', name: 'retries', type: 'number', default: 3,
        validate: { min: 0, max: 5 } },
      { displayName: 'Response', name: 'responseFormat', type: 'options', default: 'auto',
        options: [{ name: 'Auto-detect', value: 'auto' }, { name: 'JSON', value: 'json' },
                  { name: 'Text', value: 'text' }] },
      { displayName: 'Continue on failure', name: 'continueOnFail', type: 'boolean',
        default: false, hint: 'Send the error down the branch instead of stopping the workflow' },
    ],
  },

  async execute(ctx) {
    const out = [];
    const items = ctx.getInputData();

    for (let i = 0; i < items.length; i++) {
      const url = String(ctx.getNodeParameter('url', i));
      if (!/^https?:\/\//i.test(url))
        throw new WorkflowError(`URL must start with http:// or https:// — got "${url}"`,
                                { code: 'BAD_PARAMETER' });

      const method  = ctx.getNodeParameter('method', i, 'GET');
      const headers = ctx.getNodeParameter('sendHeaders', i, false)
        ? parseJsonParameter(ctx, 'headers', i) : {};
      const body    = ['POST','PUT','PATCH'].includes(method)
        ? JSON.stringify(parseJsonParameter(ctx, 'body', i)) : undefined;

      try {
        const json = await ctx.helpers.httpRequest(url, {
          method, body,
          headers: { 'content-type': 'application/json', ...headers },
          timeoutMs: ctx.getNodeParameter('timeout', i, 30000),
          retries:   ctx.getNodeParameter('retries', i, 3),
        });
        out.push({ json, pairedItem: i });        // remember which input made this output
      } catch (err) {
        if (!ctx.getNodeParameter('continueOnFail', i, false)) throw err;
        out.push({ json: { error: err.message }, pairedItem: i });
      }
    }
    return [out];
  },
};

// a JSON field the user typed is a string; a bad one must name the field, not throw "Unexpected token"
function parseJsonParameter(ctx, name, i) {
  const raw = ctx.getNodeParameter(name, i, '{}');
  if (typeof raw === 'object') return raw;
  try { return JSON.parse(raw); }
  catch { throw new WorkflowError(`The "${name}" field is not valid JSON`, { code: 'BAD_PARAMETER' }); }
}
pairedItem — the index of the input item that produced this output. Nobody needs it on day one, and everybody needs it the first time a workflow fans out to fifty items and one of them fails. Recording it costs one property.

One component turns any description into a form — and validates it

apps/editor/src/ParameterPanel.tsx 
function isVisible(p, values) {
  const { show, hide } = p.displayOptions ?? {};
  const matches = (rules) => Object.entries(rules)
    .every(([field, allowed]) => (allowed as unknown[]).includes(values[field]));
  if (show && !matches(show)) return false;
  if (hide && matches(hide))  return false;
  return true;
}

export function validateParameter(p, value): string | null {
  const isExpression = typeof value === 'string' && value.includes('{{');
  if (isExpression) return null;                       // we cannot know the value until it runs
  if (p.required && (value === '' || value == null)) return `${p.displayName} is required`;
  const v = p.validate ?? {};
  if (v.pattern && !new RegExp(v.pattern).test(String(value ?? '')))
    return v.message ?? `${p.displayName} is not in the right format`;
  if (p.type === 'number') {
    if (!Number.isFinite(Number(value))) return `${p.displayName} must be a number`;
    if (v.min != null && Number(value) < v.min) return `${p.displayName} must be at least ${v.min}`;
    if (v.max != null && Number(value) > v.max) return `${p.displayName} must be at most ${v.max}`;
  }
  if (p.type === 'json') {
    try { JSON.parse(typeof value === 'string' ? value : JSON.stringify(value)); }
    catch { return `${p.displayName} is not valid JSON`; }
  }
  return null;
}

export function ParameterPanel({ properties, values, sampleItem, onChange, onValidity }) {
  const visible = properties.filter(p => isVisible(p, values));
  const errors  = Object.fromEntries(visible
    .map(p => [p.name, validateParameter(p, values[p.name] ?? p.default)])
    .filter(([, e]) => e));

  useEffect(() => onValidity(Object.keys(errors).length === 0), [JSON.stringify(errors)]);

  return visible.map(p => {
    const v   = values[p.name] ?? p.default;
    const set = (val) => onChange({ ...values, [p.name]: val });
    const common = { id: p.name, 'aria-invalid': !!errors[p.name], value: v ?? '' };

    return (
      <Field key={p.name} label={p.displayName} hint={p.hint}
             error={errors[p.name]} preview={previewOf(v, sampleItem)}>
        {p.type === 'string'  && <input {...common} placeholder={p.placeholder}
                                     onChange={e => set(e.target.value)} />}
        {p.type === 'number'  && <input {...common} type="number"
                                     onChange={e => set(e.target.value === '' ? '' : +e.target.value)} />}
        {p.type === 'boolean' && <input id={p.name} type="checkbox" checked={!!v}
                                     onChange={e => set(e.target.checked)} />}
        {p.type === 'json'    && <textarea {...common} rows={4}
                                     onChange={e => set(e.target.value)} />}
        {p.type === 'options' && <select {...common} onChange={e => set(e.target.value)}>
                                     {p.options.map(o =>
                                       <option key={o.value} value={o.value}>{o.name}</option>)}
                                   </select>}
      </Field>
    );
  });
}

// show what an expression will actually become, using the last item that ran
function previewOf(value, sampleItem) {
  if (typeof value !== 'string' || !value.includes('{{') || !sampleItem) return null;
  try { return String(resolve(value, sampleItem, {})); }
  catch (e) { return `⚠ ${e.message}`; }
}

Disable the Run button while onValidity is false. A workflow that cannot possibly work should never reach the engine — the user finds out in the sidebar, in red, in the field that is wrong.

Expressions, resolved per item — and unable to reach the server

packages/engine/expressions.js 
import { WorkflowError } from './errors.js';

// names an expression must never see. They are passed in as parameters with no
// arguments, so inside the expression each one is simply `undefined`.
const SHADOWED = ['process', 'require', 'module', 'exports', 'globalThis', 'global',
                  'fetch', 'eval', 'Function', 'setTimeout', 'setInterval',
                  'XMLHttpRequest', 'WebSocket', 'import'];

const cache = new Map();                     // compiling is the slow part; do it once per string

function compile(expr) {
  if (cache.has(expr)) return cache.get(expr);
  if (expr.length > 2000) throw new WorkflowError('That expression is too long');
  let fn;
  try {
    fn = new Function('$json', '$node', '$now', '$itemIndex', ...SHADOWED,
                      `"use strict"; return (${expr});`);
  } catch (e) {
    throw new WorkflowError(`Expression will not compile: ${e.message}`, { code: 'BAD_EXPRESSION' });
  }
  if (cache.size > 500) cache.clear();
  cache.set(expr, fn);
  return fn;
}

export function resolve(value, item, results, itemIndex = 0) {
  if (typeof value !== 'string' || !value.includes('{{')) return value;

  const evaluate = (expr) => {
    const $json = item?.json ?? {};
    const $node = new Proxy({}, {
      get: (_, name) => {
        const branches = results[name];
        if (!branches) throw new WorkflowError(`$node["${String(name)}"] has not run yet`,
                                               { code: 'BAD_EXPRESSION' });
        return { json: branches[0]?.[0]?.json ?? {} };
      },
    });
    try {
      return compile(expr.trim())($json, $node, new Date(), itemIndex);
    } catch (e) {
      if (e instanceof WorkflowError) throw e;
      throw new WorkflowError(`{{${expr.trim()}}} → ${e.message}`, { code: 'BAD_EXPRESSION' });
    }
  };

  const whole = value.match(/^\s*\{\{([\s\S]+)\}\}\s*$/);   // the field is ONE expression
  if (whole) return evaluate(whole[1]);                      // …so keep the real type
  return value.replace(/\{\{([\s\S]+?)\}\}/g, (_, e) => String(evaluate(e) ?? ''));
}

Now wire it into the engine. This is the whole change to §2.1b — three lines, and from this moment every setting of every node supports expressions, forever:

getNodeParameter(name, i = 0, fallback = undefined) {
  const raw = node.parameters?.[name] ?? fallback;
  if (raw === undefined)
    throw new WorkflowError(`"${nodeName}" needs its "${name}" setting filled in`,
